1. 用户与组织域

1.1 organization — 组织架构表

-- 组织架构树，支持自引用父子关系，使用 adjacency list 模式 CREATE TABLE organization ( id BIGINT NOT NULL AUTO_INCREMENT COMMENT '组织ID', parent_id BIGINT DEFAULT NULL COMMENT '上级组织ID，根节点为NULL', org_code VARCHAR(64) NOT NULL COMMENT '组织编码，全系统唯一', org_name VARCHAR(128) NOT NULL COMMENT '组织名称', org_type TINYINT NOT NULL DEFAULT 1 COMMENT '类型: 1=公司 2=部门 3=小组', sort_order INT NOT NULL DEFAULT 0 COMMENT '同级排序号', path VARCHAR(512) DEFAULT NULL COMMENT '物化路径: /1/5/12/', status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0=禁用 1=启用', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_org_code (org_code), KEY idx_parent_id (parent_id), KEY idx_path (path(255)) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='组织架构表';
字段类型约束/索引说明
idBIGINTPK AUTO_INCREMENT组织ID
parent_idBIGINTFK→self · IDX自引用外键，根节点 NULL
org_codeVARCHAR(64)UNIQ业务编码，唯一
pathVARCHAR(512)IDX 前缀255物化路径，快速查子树
org_typeTINYINTDEFAULT 11=公司 2=部门 3=小组
statusTINYINTDEFAULT 10=禁用 1=启用
设计要点：path 物化路径用于高效查询子树（WHERE path LIKE '/1/5/%'），避免递归 CTE 在深层级下的性能问题。同时保留 parent_id 用于直接父子查询。

1.2 user — 用户表

-- 用户账号与基本信息，密码使用 bcrypt 哈希存储 CREATE TABLE user ( id BIGINT NOT NULL AUTO_INCREMENT COMMENT '用户ID', org_id BIGINT NOT NULL COMMENT '所属组织ID', username VARCHAR(64) NOT NULL COMMENT '登录用户名', password_hash VARCHAR(255) NOT NULL COMMENT 'bcrypt哈希', real_name VARCHAR(64) NOT NULL COMMENT '真实姓名', email VARCHAR(128) DEFAULT NULL COMMENT '邮箱', phone VARCHAR(32) DEFAULT NULL COMMENT '手机号', avatar_url VARCHAR(512) DEFAULT NULL COMMENT '头像URL', employee_no VARCHAR(64) DEFAULT NULL COMMENT '工号', position VARCHAR(128) DEFAULT NULL COMMENT '职务', status TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 0=禁用 1=启用 2=锁定', last_login_at DATETIME DEFAULT NULL COMMENT '最后登录时间', last_login_ip VARCHAR(45) DEFAULT NULL COMMENT '最后登录IP', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_username (username), UNIQUE KEY uk_email (email), UNIQUE KEY uk_employee_no (employee_no), KEY idx_org_id (org_id), KEY idx_phone (phone), CONSTRAINT fk_user_org FOREIGN KEY (org_id) REFERENCES organization(id) ON DELETE RESTRICT ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';
字段类型约束/索引说明
idBIGINTPK用户ID
org_idBIGINTFK→organization · IDX所属组织，禁止删除有用户的组织
usernameVARCHAR(64)UNIQ登录用户名
emailVARCHAR(128)UNIQ邮箱，可空但非空时唯一
employee_noVARCHAR(64)UNIQ工号，可空但非空时唯一
statusTINYINTDEFAULT 10=禁用 1=启用 2=锁定（连续登录失败）

1.3 role — 角色表

-- RBAC 角色定义，角色与权限多对多 CREATE TABLE role ( id BIGINT NOT NULL AUTO_INCREMENT, role_code VARCHAR(64) NOT NULL COMMENT '角色编码: ADMIN, MANAGER, APPROVER...', role_name VARCHAR(128) NOT NULL COMMENT '角色名称', description VARCHAR(256) DEFAULT NULL, is_system TINYINT NOT NULL DEFAULT 0 COMMENT '是否系统内置角色(不可删)', sort_order INT NOT NULL DEFAULT 0, status TINYINT NOT NULL DEFAULT 1, created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_role_code (role_code) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='角色表';

1.4 user_role — 用户角色关联表

-- 用户与角色多对多关联，中间表 CREATE TABLE user_role ( id BIGINT NOT NULL AUTO_INCREMENT, user_id BIGINT NOT NULL, role_id BIGINT NOT NULL, granted_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_user_role (user_id, role_id), KEY idx_role_id (role_id), CONSTRAINT fk_ur_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE, CONSTRAINT fk_ur_role FOREIGN KEY (role_id) REFERENCES role(id) ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户角色关联表';
外键策略：用户/角色删除时级联删除关联记录。uk_user_role 联合唯一索引防止重复授权。

2. 流程定义域

2.1 process_definition — 流程定义表

-- 流程模板蓝图，支持版本管理，JSON 存储节点与连线拓扑 CREATE TABLE process_definition ( id BIGINT NOT NULL AUTO_INCREMENT, def_key VARCHAR(64) NOT NULL COMMENT '流程唯一标识(同一key多版本)', name VARCHAR(128) NOT NULL COMMENT '流程名称', version INT NOT NULL DEFAULT 1 COMMENT '版本号，每次发布+1', category VARCHAR(64) DEFAULT NULL COMMENT '分类: 报销/请假/公文...', form_config JSON DEFAULT NULL COMMENT '表单字段定义JSON', node_json JSON NOT NULL COMMENT '节点拓扑JSON', status TINYINT NOT NULL DEFAULT 0 COMMENT '0=草稿 1=已发布 2=已停用', creator_id BIGINT NOT NULL, published_at DATETIME DEFAULT NULL COMMENT '发布时间', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_def_key_version (def_key, version), KEY idx_category (category), KEY idx_status (status), CONSTRAINT fk_pd_creator FOREIGN KEY (creator_id) REFERENCES user(id) ON DELETE RESTRICT ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='流程定义表';
node_json 结构示例：[{"id":"n1","type":"START","name":"开始","next":"n2"},{"id":"n2","type":"APPROVAL","name":"经理审批","assigneeType":"ROLE","assigneeValue":"MANAGER","mode":"SINGLE","next":"n3"},{"id":"n3","type":"END","name":"结束"}]

2.2 node_definition — 节点定义表

-- 从 node_json 中抽出的结构化节点定义，便于查询和统计 CREATE TABLE node_definition ( id BIGINT NOT NULL AUTO_INCREMENT, def_id BIGINT NOT NULL COMMENT '所属流程定义ID', node_key VARCHAR(64) NOT NULL COMMENT '节点标识(n1,n2...)', node_name VARCHAR(128) NOT NULL COMMENT '节点名称', node_type VARCHAR(32) NOT NULL COMMENT 'START|APPROVAL|CONDITION|PARALLEL|END', approval_mode VARCHAR(32) DEFAULT NULL COMMENT 'SINGLE|COUNTERSIGN|OR_SIGN', assignee_type VARCHAR(32) DEFAULT NULL COMMENT 'USER|ROLE|DEPT_LEADER|INITIATOR_LEADER|EXPR', assignee_value VARCHAR(512) DEFAULT NULL COMMENT '审批人标识: userId/roleCode/表达式', condition_expr VARCHAR(512) DEFAULT NULL COMMENT '条件表达式(条件节点)', sort_order INT NOT NULL DEFAULT 0, PRIMARY KEY (id), UNIQUE KEY uk_def_node (def_id, node_key), KEY idx_node_type (node_type), CONSTRAINT fk_nd_def FOREIGN KEY (def_id) REFERENCES process_definition(id) ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='节点定义表';

3. 流程实例域

3.1 process_instance — 流程实例表

-- 一次具体的审批申请，核心枢纽表 CREATE TABLE process_instance ( id BIGINT NOT NULL AUTO_INCREMENT, instance_no VARCHAR(64) NOT NULL COMMENT '流水号: OA20260829001', def_id BIGINT NOT NULL COMMENT '流程定义ID', def_version INT NOT NULL COMMENT '快照: 发起时的定义版本', title VARCHAR(256) NOT NULL COMMENT '申请标题', initiator_id BIGINT NOT NULL COMMENT '发起人ID', business_type VARCHAR(32) NOT NULL COMMENT '业务类型: REIMBURSEMENT/LEAVE/DOCUMENT...', business_id BIGINT DEFAULT NULL COMMENT '关联业务表ID(document.id等)', business_data JSON DEFAULT NULL COMMENT '表单数据快照', current_node VARCHAR(64) DEFAULT NULL COMMENT '当前节点key', status TINYINT NOT NULL DEFAULT 0 COMMENT '0=草稿 1=进行中 2=已完成 3=已撤销', priority TINYINT NOT NULL DEFAULT 0 COMMENT '优先级: 0=普通 1=紧急 2=特急', submitted_at DATETIME DEFAULT NULL COMMENT '提交时间', completed_at DATETIME DEFAULT NULL COMMENT '完结时间', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_instance_no (instance_no), KEY idx_def_id (def_id), KEY idx_initiator (initiator_id), KEY idx_status (status), KEY idx_business (business_type, business_id), KEY idx_created (created_at), CONSTRAINT fk_pi_def FOREIGN KEY (def_id) REFERENCES process_definition(id), CONSTRAINT fk_pi_user FOREIGN KEY (initiator_id) REFERENCES user(id) ON DELETE RESTRICT ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='流程实例表';
字段类型约束/索引说明
instance_noVARCHAR(64)UNIQ业务流水号，便于人类识别
def_idBIGINTFK · IDX流程定义ID
initiator_idBIGINTFK · IDX发起人，禁止删除有实例的用户
business_dataJSON—表单数据快照，审批时冻结
statusTINYINTIDX0=草稿 1=进行中 2=已完成 3=已撤销
business_type + business_id组合IDX多态关联业务表

3.2 task — 任务表

-- 审批待办任务，每个审批节点生成一条 CREATE TABLE task ( id BIGINT NOT NULL AUTO_INCREMENT, instance_id BIGINT NOT NULL COMMENT '流程实例ID', node_key VARCHAR(64) NOT NULL COMMENT '节点标识', node_name VARCHAR(128) NOT NULL COMMENT '节点名称(快照)', assignee_id BIGINT DEFAULT NULL COMMENT '实际处理人ID', delegate_from BIGINT DEFAULT NULL COMMENT '转办来源人ID(转办场景)', task_type VARCHAR(32) NOT NULL DEFAULT 'APPROVAL' COMMENT 'APPROVAL|CC|DELEGATE', status TINYINT NOT NULL DEFAULT 0 COMMENT '0=待处理 1=处理中 2=已通过 3=已驳回 4=已转办', comment TEXT DEFAULT NULL COMMENT '审批意见', due_at DATETIME DEFAULT NULL COMMENT '截止时间(超时提醒)', claimed_at DATETIME DEFAULT NULL COMMENT '签收时间', completed_at DATETIME DEFAULT NULL COMMENT '完成时间', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY idx_instance (instance_id), KEY idx_assignee_status (assignee_id, status), KEY idx_status (status), KEY idx_due (due_at), CONSTRAINT fk_t_instance FOREIGN KEY (instance_id) REFERENCES process_instance(id) ON DELETE CASCADE, CONSTRAINT fk_t_assignee FOREIGN KEY (assignee_id) REFERENCES user(id) ON DELETE SET NULL ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='任务表';
关键索引：idx_assignee_status (assignee_id, status) 覆盖「我的待办」最高频查询：WHERE assignee_id=? AND status=0。idx_due 支撑超时巡检定时任务。

3.3 task_candidate — 任务候选人表

-- 会签/或签场景下，一个节点对应多个候选人 CREATE TABLE task_candidate ( id BIGINT NOT NULL AUTO_INCREMENT, task_id BIGINT NOT NULL, user_id BIGINT NOT NULL, PRIMARY KEY (id), UNIQUE KEY uk_task_user (task_id, user_id), KEY idx_user (user_id), CONSTRAINT fk_tc_task FOREIGN KEY (task_id) REFERENCES task(id) ON DELETE CASCADE, CONSTRAINT fk_tc_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='任务候选人表';

4. 业务域

4.1 document — 公文表

-- 公文正文与元数据，关联流程实例 CREATE TABLE document ( id BIGINT NOT NULL AUTO_INCREMENT, doc_no VARCHAR(64) DEFAULT NULL COMMENT '公文文号', title VARCHAR(256) NOT NULL COMMENT '公文标题', content LONGTEXT DEFAULT NULL COMMENT '正文内容(HTML)', doc_type VARCHAR(32) NOT NULL COMMENT '类型: NOTICE/REPORT/INSTRUCTION/DECISION', urgency TINYINT NOT NULL DEFAULT 0 COMMENT '紧急度: 0=普通 1=急 2=特急', secrecy_level TINYINT NOT NULL DEFAULT 0 COMMENT '密级: 0=公开 1=内部 2=机密', author_id BIGINT NOT NULL COMMENT '拟稿人ID', instance_id BIGINT DEFAULT NULL COMMENT '关联流程实例ID', status TINYINT NOT NULL DEFAULT 0 COMMENT '0=草稿 1=审批中 2=已发布 3=已归档', published_at DATETIME DEFAULT NULL, archived_at DATETIME DEFAULT NULL, created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, PRIMARY KEY (id), UNIQUE KEY uk_doc_no (doc_no), KEY idx_author (author_id), KEY idx_instance (instance_id), KEY idx_status (status), KEY idx_type_urgency (doc_type, urgency), FULLTEXT KEY ft_title_content (title, content), CONSTRAINT fk_doc_author FOREIGN KEY (author_id) REFERENCES user(id) ON DELETE RESTRICT, CONSTRAINT fk_doc_instance FOREIGN KEY (instance_id) REFERENCES process_instance(id) ON DELETE SET NULL ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='公文表';
FULLTEXT 全文索引支持公文标题和正文的中文全文检索。InnoDB 全文索引需 ngram 分词器，配置 innodb_ft_min_token_size=2。

4.2 attachment — 附件表

-- 公文/流程实例的附件，文件存储在对象存储(OSS/MinIO) CREATE TABLE attachment ( id BIGINT NOT NULL AUTO_INCREMENT, target_type VARCHAR(32) NOT NULL COMMENT '关联类型: DOCUMENT|INSTANCE|TASK', target_id BIGINT NOT NULL COMMENT '关联目标ID', file_name VARCHAR(256) NOT NULL COMMENT '原始文件名', file_size BIGINT NOT NULL COMMENT '文件大小(字节)', file_type VARCHAR(64) NOT NULL COMMENT 'MIME类型', storage_key VARCHAR(512) NOT NULL COMMENT '对象存储key', storage_url VARCHAR(1024) DEFAULT NULL COMMENT '访问URL(可预签名)', uploader_id BIGINT NOT NULL, created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY idx_target (target_type, target_id), CONSTRAINT fk_att_uploader FOREIGN KEY (uploader_id) REFERENCES user(id) ON DELETE RESTRICT ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='附件表';

4.3 approval_record — 审批记录表

-- 审批操作留痕，独立于task表，确保历史不被重分配丢失 CREATE TABLE approval_record ( id BIGINT NOT NULL AUTO_INCREMENT, instance_id BIGINT NOT NULL COMMENT '流程实例ID', task_id BIGINT DEFAULT NULL COMMENT '关联任务ID(可空:发起操作无任务)', node_key VARCHAR(64) NOT NULL COMMENT '节点标识', action VARCHAR(32) NOT NULL COMMENT 'SUBMIT|APPROVE|REJECT|TRANSFER|CANCEL|COUNTERSIGN', operator_id BIGINT NOT NULL COMMENT '操作人ID', operator_name VARCHAR(64) NOT NULL COMMENT '操作人姓名快照', operator_dept VARCHAR(128) DEFAULT NULL COMMENT '操作时部门快照', comment TEXT DEFAULT NULL COMMENT '审批意见', from_node VARCHAR(64) DEFAULT NULL COMMENT '驳回来源节点(驳回时)', to_node VARCHAR(64) DEFAULT NULL '目标节点(驳回/转办时)', created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY idx_instance (instance_id), KEY idx_task (task_id), KEY idx_operator (operator_id), CONSTRAINT fk_ar_instance FOREIGN KEY (instance_id) REFERENCES process_instance(id) ON DELETE CASCADE, CONSTRAINT fk_ar_task FOREIGN KEY (task_id) REFERENCES task(id) ON DELETE SET NULL, CONSTRAINT fk_ar_operator FOREIGN KEY (operator_id) REFERENCES user(id) ON DELETE RESTRICT ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='审批记录表';
设计要点：operator_name 和 operator_dept 做了快照冗余——即使用户后续改名或调岗，审批历史仍保留操作时的信息。task_id 使用 ON DELETE SET NULL 而非 CASCADE，确保任务删除不丢失审批记录。

5. 消息通知域

5.1 notification — 消息通知表

-- 站内消息通知，支持多种渠道推送 CREATE TABLE notification ( id BIGINT NOT NULL AUTO_INCREMENT, user_id BIGINT NOT NULL COMMENT '接收人ID', title VARCHAR(256) NOT NULL, content TEXT NOT NULL, notify_type VARCHAR(32) NOT NULL COMMENT 'TASK|APPROVAL|SYSTEM|REMINDER', ref_type VARCHAR(32) DEFAULT NULL COMMENT '关联类型', ref_id BIGINT DEFAULT NULL COMMENT '关联ID', is_read TINYINT NOT NULL DEFAULT 0 COMMENT '0=未读 1=已读', read_at DATETIME DEFAULT NULL, created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP, PRIMARY KEY (id), KEY idx_user_read (user_id, is_read), KEY idx_user_created (user_id, created_at), KEY idx_type (notify_type), CONSTRAINT fk_notif_user FOREIGN KEY (user_id) REFERENCES user(id) ON DELETE CASCADE ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='消息通知表';
idx_user_read (user_id, is_read) 覆盖「未读消息数」查询：SELECT COUNT(*) FROM notification WHERE user_id=? AND is_read=0。

外键onDelete 策略理由
user.org_id → organizationRESTRICT禁止删除有用户的组织
